const fs = require('node:fs');
const assert = require('node:assert/strict');
const { PNG } = require('/Users/manh.pd1/.cache/codex-runtimes/codex-primary-runtime/dependencies/node/node_modules/pngjs');
const root = process.argv[2] ?? '/tmp/tabula-issue84-renderer/evidence';
const upright = [[240,30,40],[30,200,50],[40,80,230],[240,190,20]];
const flipped = [upright[2],upright[3],upright[0],upright[1]];
const scopes = [['unbounded',30],['clipped',280],['nested',520],['reset',750]];
const pixel = (png,x,y) => [...png.data.subarray((y*png.width+x)*4,(y*png.width+x)*4+3)];
let cornerSets=0,inkComparisons=0;
for (const negative of [false,true]) for (const dpr of [1,2]) {
  const png = PNG.sync.read(fs.readFileSync(`${root}/${negative?'negative':'actual'}-macroquad-clips-dpr${dpr}.png`));
  assert.equal(png.width,960*dpr); assert.equal(png.height,360*dpr);
  for (const [scope,x] of scopes) {
    const samples = [[x+20,75],[x+140,75],[x+20,155],[x+140,155]].map(([px,py])=>pixel(png,px*dpr,py*dpr));
    assert.deepEqual(samples,negative && ['clipped','nested'].includes(scope)?flipped:upright,`${scope}, negative=${negative}, DPR${dpr}`);
    cornerSets++;
  }
  if (!negative) {
    for(let y=180*dpr;y<220*dpr;y++) for(let x=0;x<180*dpr;x++) {
      assert.deepEqual(pixel(png,30*dpr+x,y),pixel(png,280*dpr+x,y),`Vietnamese ink x=${x},y=${y},DPR${dpr}`);
    }
    inkComparisons++;
  }
}
console.log(JSON.stringify({status:'PASS',corner_sets:cornerSets,current_corner_sets:8,negative_corner_sets:8,ink_comparisons:inkComparisons,ink_strip:{first_origin:[30,180],second_origin:[280,180],logical_size:[180,40]},corner_samples:'x+20/x+140 at y75/y155; RGB equality'}));
