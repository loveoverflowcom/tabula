use glam::Vec2;
use macroquad::prelude as mq;
use tabula_design::{Color, Theme, ThemeKind, TextStyleToken};
use tabula_presentation::{Align, AssetRef, Camera2D, Corners, Dpi, Layer, Paint, Rect, RenderCmd, RenderListBuilder, Renderer, Viewport};
use tabula_render_macroquad::MacroquadRenderer;
use tabula_assets::{AssetPackManifest,UnverifiedAssetBytes};
use image::ImageEncoder;
fn conf()->mq::Conf{mq::Conf{window_title:"Issue84 actual renderer clip fixture".into(),window_width:960,window_height:720,high_dpi:true,..Default::default()}}
fn text(b:&mut RenderListBuilder,t:&str,x:f32,y:f32){b.push(RenderCmd::Text{text:t.into(),at:Vec2::new(x,y),style:TextStyleToken::BodyLg,align:Align::Start,max_width:None,color:Color::rgba(0,0,0,255),layer:Layer::BOARD,z:0}).unwrap();}
fn sprite(b:&mut RenderListBuilder,x:f32,y:f32){b.push(RenderCmd::Sprite{asset:AssetRef::from_static("corners"),rect:Rect::new(Vec2::new(x,y),Vec2::new(160.0,120.0)).unwrap(),tint:Color::rgba(255,255,255,255),rotation:0.0,pivot:Vec2::ZERO,layer:Layer::BOARD,z:0}).unwrap();}
fn clip(b:&mut RenderListBuilder,x:f32,y:f32,w:f32,h:f32){b.push(RenderCmd::PushClip{rect:Rect::new(Vec2::new(x,y),Vec2::new(w,h)).unwrap(),layer:Layer::BOARD,z:0}).unwrap();}
fn pop(b:&mut RenderListBuilder){b.push(RenderCmd::PopClip{layer:Layer::BOARD,z:0}).unwrap();}
#[macroquad::main(conf)]

 let mut renderer=MacroquadRenderer::new();
 renderer.set_builtin_font_bytes(include_bytes!("/Users/manh.pd1/Projects/Mine/tabula/apps/game-client/web/assets/OpenSans-Regular.ttf"),include_bytes!("/Users/manh.pd1/Projects/Mine/tabula/apps/game-client/web/assets/OpenSans-Semibold.ttf"),include_bytes!("/Users/manh.pd1/Projects/Mine/tabula/apps/game-client/web/assets/NotoSerif-Bold.ttf")).unwrap();
 let mut pixels=Vec::new(); for y in 0..12 {for x in 0..16 { pixels.extend_from_slice(match(x<8,y<6){(true,true)=>&[240,30,40,255],(false,true)=>&[30,200,50,255],(true,false)=>&[40,80,230,255],(false,false)=>&[240,190,20,255]}); }}
 let mut png=Vec::new();image::codecs::png::PngEncoder::new(&mut png).write_image(&pixels,16,12,image::ColorType::Rgba8).unwrap();
 let manifest=AssetPackManifest::from_toml(&format!(r#"pack = "fixture"
version = "1.0.0"

 renderer.assets_mut().bind_pack(&manifest,manifest.game(),manifest.pack_ref()).unwrap();
